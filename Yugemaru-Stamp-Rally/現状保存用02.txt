        function renderSchedule() {
            const list = document.getElementById('schedule-list');

            list.innerHTML = '';

                list.innerHTML = '<p class="text-sm text-marine-600">現在スケジュールはありません。</p>';
                return;

            const currentHours = now.getHours();
            const currentMinutes = now.getMinutes();
            const currentTimeStr = `${String(currentHours).padStart(2, '0')}:${String(currentMinutes).padStart(2, '0')}`;

            // 現在のスケジュールインデックスを特定
            let currentIndex = 0;
            for (let i = 0; i < CONFIG.schedule.length; i++) {
                if (currentTimeStr >= CONFIG.schedule[i].time) {
                    currentIndex = i;
                } else {
                    break;
                }
            }

            CONFIG.schedule.forEach((item, index) => {
                const isCurrent = index === currentIndex;
                const isLast = index === CONFIG.schedule.length - 1;
                const li = document.createElement('li');
                li.className = isLast ? 'ml-6' : 'mb-6 ml-6';
                
                // 現在の予定の丸ポッチをオレンジ色で点滅させる

                // 現在の予定のテキストをハイライト

                const timeClass = isCurrent ? 'text-accent-500' : 'text-marine-900';

        async function startScanner() {
            document.getElementById('screen-scanner').classList.remove('hidden');
            html5QrCode = new Html5Qrcode("reader");
            const config = { fps: 10, qrbox: { width: 250, height: 250 }, aspectRatio: 1.0 };
            try {
                await html5QrCode.start({ facingMode: "environment" }, config, onScanSuccess, onScanFailure);
                isScannerRunning = true;
            } catch (err) {
                console.error(err);
                stopScanner();
                showModal('カメラエラー', 'カメラの起動に失敗しました。\nブラウザの設定を確認してください。', '📷');
            }
        }

        async function stopScanner() {
            if (html5QrCode && isScannerRunning) {
                try { await html5QrCode.stop(); html5QrCode.clear(); } catch (e) {}
                isScannerRunning = false;
            }
            document.getElementById('screen-scanner').classList.add('hidden');
        }

        function onScanSuccess(decodedText) {
            stopScanner().then(() => processStamp(decodedText));
        }
        function onScanFailure() {}

        // --- スタンプ処理 ---
        async function processStamp(scannedId) {
            const id = scannedId.trim();
            const activeSpots = getActiveSpots();
            const spot = activeSpots.find(s => s.id === id);
            
            if (!spot) { showModal('エラー', '無効なQRコード、または現在のコースに含まれていないスポットです。', '❓'); return; }
            if (acquiredStamps.includes(id)) { showModal('お知らせ', '取得済みです！', spot.icon); return; }

            switchView('stamp');
            await playStampAnimation(spot);

            acquiredStamps.push(id);
            localStorage.setItem(CONFIG.storageKey, JSON.stringify(acquiredStamps));
            updateUI();

            setTimeout(() => {
                if (acquiredStamps.length >= activeSpots.length) {
                    triggerComplete();
                } else {
                    fireConfetti();
                    showModal('スタンプGET！', `${spot.name}のスタンプを獲得しました！\n\n${spot.description}`, spot.icon);
                }
            }, 1000); // アニメーション終了から少し早めにポップアップ表示
        }

        // --- 演出関連 ---
        // 通常のスタンプ獲得時の紙吹雪
        function fireConfetti() {
            const count = 200;
            const defaults = {
                origin: { y: 0.6 },
                zIndex: 150
            };

            function fire(particleRatio, opts) {
                confetti(Object.assign({}, defaults, opts, {
                    particleCount: Math.floor(count * particleRatio)
                }));
            }

            fire(0.25, { spread: 26, startVelocity: 55 });
            fire(0.2, { spread: 60 });
            fire(0.35, { spread: 100, decay: 0.91, scalar: 0.8 });
            fire(0.1, { spread: 120, startVelocity: 25, decay: 0.92, scalar: 1.2 });
            fire(0.1, { spread: 120, startVelocity: 45 });
        }

        function playStampAnimation(spot) {
            return new Promise(resolve => {
                const cell = document.getElementById(`stamp-cell-${spot.id}`);
                if (!cell) { resolve(); return; }

                cell.scrollIntoView({ behavior: 'smooth', block: 'center' });

                setTimeout(() => {
                    const stampEl = document.createElement('img');
                    stampEl.src = CONFIG.images.stamp;
                    stampEl.className = `absolute -top-1 -right-1 w-16 h-16 md:w-20 md:h-20 object-contain z-30 animate-stamp-press drop-shadow-md`;
                    
                    stampEl.onerror = function() {
                        this.onerror = null;
                        this.src = 'https://placehold.co/200x200/transparent/94a3b8?text=Stamp';
                    };
                    cell.appendChild(stampEl);

                    setTimeout(() => { resolve(); }, 600);
                }, 300);
            });
        }

        // --- コース変更機能 ---
        function setCourse(courseId) {
            if (currentCourseId === courseId) return;
            
            // 誤タップ防止のため、1回目のタップで確認メッセージを表示
            if (pendingCourseChange !== courseId) {
                pendingCourseChange = courseId;
                updateCourseUI(); // 一旦UIをリセット
                
                const btn = document.getElementById(`btn-course-${courseId}`);
                btn.innerHTML = `<span class="text-red-500 font-bold text-sm mx-auto">本当に変更しますか？<br>(タップで確定/リセット)</span>`;
                btn.className = "border-2 border-red-500 bg-red-50 text-marine-900 py-3 rounded-xl font-bold flex justify-center px-4 items-center";
                
                // 4秒放置でキャンセル
                setTimeout(() => {
                    if (pendingCourseChange === courseId) {
                        pendingCourseChange = null;
                        updateCourseUI();
                    }
                }, 4000);
                return;
            }

            // 2回目のタップで確定処理
            pendingCourseChange = null;
            currentCourseId = courseId;
            localStorage.setItem(CONFIG.storageKey + "_course", currentCourseId);
            
            // スタンプ状態をリセット
            acquiredStamps = []; 
            localStorage.setItem(CONFIG.storageKey, JSON.stringify(acquiredStamps));
            
            document.getElementById('stamp-grid').innerHTML = ''; // グリッド枠を再生成させる
            updateCourseUI();
            updateUI();
            generateTestQRs();
        }

        function updateCourseUI() {
            const btnSunny = document.getElementById('btn-course-sunny');
            const btnRainy = document.getElementById('btn-course-rainy');
            
            const sunnyText = `${customCourses.sunny.name} (${customCourses.sunny.spots.length}カ所)`;
            const rainyText = `${customCourses.rainy.name} (${customCourses.rainy.spots.length}カ所)`;
            
            // テキストを状態に戻す
            btnSunny.innerHTML = `<span>${sunnyText}</span><span class="check-icon hidden text-xl">✅</span>`;
            btnRainy.innerHTML = `<span>${rainyText}</span><span class="check-icon hidden text-xl">✅</span>`;
            
            if (currentCourseId === 'sunny') {
                btnSunny.className = "border-2 border-accent-500 bg-accent-50 text-marine-900 py-3 rounded-xl font-bold flex justify-between px-4 items-center transition-colors";
                btnSunny.querySelector('.check-icon').classList.remove('hidden');
                
                btnRainy.className = "border-2 border-gray-200 hover:bg-gray-50 text-gray-500 py-3 rounded-xl font-bold flex justify-between px-4 items-center transition-colors cursor-pointer";
                btnRainy.querySelector('.check-icon').classList.add('hidden');
            } else {
                btnRainy.className = "border-2 border-blue-500 bg-blue-50 text-marine-900 py-3 rounded-xl font-bold flex justify-between px-4 items-center transition-colors";
                btnRainy.querySelector('.check-icon').classList.remove('hidden');
                
                btnSunny.className = "border-2 border-gray-200 hover:bg-gray-50 text-gray-500 py-3 rounded-xl font-bold flex justify-between px-4 items-center transition-colors cursor-pointer";
                btnSunny.querySelector('.check-icon').classList.add('hidden');
            }
        }

        // --- コーススポット編集機能 ---
        function openCourseEditor(courseId) {
            editingCourseId = courseId;
            const isSunny = courseId === 'sunny';
            document.getElementById('course-editor-title').innerText = isSunny ? "☀️ 晴れの日コースの編集" : "☔️ 雨の日コースの編集";
            document.getElementById('course-editor-error').classList.add('hidden');
            
            const listContainer = document.getElementById('course-editor-list');
            listContainer.innerHTML = '';
            
            const activeIds = customCourses[courseId].spots;
            
            ALL_SPOTS.forEach(spot => {
                const isChecked = activeIds.includes(spot.id);
                const label = document.createElement('label');
                label.className = "flex items-center gap-3 p-3 border border-marine-100 rounded-xl hover:bg-marine-50 cursor-pointer transition-colors";
                label.innerHTML = `
                    <input type="checkbox" value="${spot.id}" class="w-6 h-6 accent-accent-500 cursor-pointer course-spot-checkbox" ${isChecked ? 'checked' : ''}>
                    <span class="text-2xl">${spot.icon}</span>
                    <div class="flex-1">
                        <div class="font-bold text-marine-900 text-sm">${spot.name}</div>
                    </div>
                `;
                listContainer.appendChild(label);
            });

            const overlay = document.getElementById('course-editor-overlay');
            overlay.classList.remove('hidden');
            setTimeout(() => {
                overlay.classList.remove('opacity-0');
                document.getElementById('course-editor-content').classList.remove('scale-95');
            }, 10);
        }

        function closeCourseEditor() {
            const overlay = document.getElementById('course-editor-overlay');
            overlay.classList.add('opacity-0'); 
            document.getElementById('course-editor-content').classList.add('scale-95');
            setTimeout(() => overlay.classList.add('hidden'), 300);
            editingCourseId = null;
        }

        function saveCourseEditor() {
            const checkboxes = document.querySelectorAll('.course-spot-checkbox:checked');
            const selectedIds = Array.from(checkboxes).map(cb => cb.value);
            
            if (selectedIds.length === 0) {
                document.getElementById('course-editor-error').classList.remove('hidden');
                const content = document.getElementById('course-editor-content');
                content.classList.add('animate-screen-shake');
                setTimeout(() => content.classList.remove('animate-screen-shake'), 350);
                return;
            }
            
            customCourses[editingCourseId].spots = selectedIds;
            localStorage.setItem(CONFIG.storageKey + "_custom_courses", JSON.stringify(customCourses));
            
            // 編集したコースが現在選択中のコースなら、スタンプをリセットしてUIを更新
            if (editingCourseId === currentCourseId) {
                acquiredStamps = [];
                localStorage.setItem(CONFIG.storageKey, JSON.stringify(acquiredStamps));
                document.getElementById('stamp-grid').innerHTML = '';
                updateUI();
            }
            
            updateCourseUI();
            closeCourseEditor();
        }

        window.addEventListener('DOMContentLoaded', init);
    </script>
